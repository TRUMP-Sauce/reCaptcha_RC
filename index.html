第三方接入代码

<!-- 引入 CSS 和 SDK -->
<link rel="stylesheet" href="css/style.css">
<script src="js/captcha-sdk.js"></script>
<div id="captcha-box"></div>

<script>
// 初始化验证码
const captcha = new NatawaykVerify({
    container: '#captcha-box',
    type: 'slider',
    difficulty: 'normal',
    onSuccess: (token) => {
        console.log('验证通过, token:', token);
        // 将 token 随表单一起提交
        document.getElementById('verify_token').value = token;
    }
});
captcha.render();

// 提交表单时验证
document.getElementById('myForm').addEventListener('submit', async (e) => {
    e.preventDefault();
    const result = await captcha.verify();
    if (result.success) {
        // 提交表单...
    }
});
</script>

后端 API 验证

// PHP 后端验证示例
$token = $_POST['verify_token'];
$ch = curl_init();
curl_setopt($ch, CURLOPT_URL, 'https://your-server/api/verify.php');
curl_setopt($ch, CURLOPT_POST, true);
curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode([
    'session_id' => $token
]));
curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
$response = curl_exec($ch);
$result = json_decode($response, true);

if ($result['code'] === 200) {
    // 验证通过，处理业务逻辑
} else {
    // 验证失败
}
